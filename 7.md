```python

numero = int(input("Digite um número inteiro: "))

# Verificar se é positivo, negativo ou nulo
if numero > 0:
    situacao = "positivo"
elif numero < 0:
    situacao = "negativo"
else:
    situacao = "nulo"

# Verificar se é par ou ímpar
if numero % 2 == 0:
    paridade = "par"
else:
    paridade = "ímpar"

# Exibir resultado
print(f"O número {numero} é {situacao} e {paridade}.")
