temperaturas = []

for i in range(7):
    temperatura = float(input(f"Digite a temperatura do {i + 1}º dia: "))
    temperaturas.append(temperatura)

maior = max(temperaturas)
menor = min(temperaturas)
media = sum(temperaturas) / len(temperaturas)

dias_acima_media = 0

for temperatura in temperaturas:
    if temperatura > media:
        dias_acima_media += 1

print("\n===== RELATÓRIO METEOROLÓGICO =====")

print("Temperaturas registradas:", temperaturas)
print(f"Maior temperatura: {maior:.2f} °C")
print(f"Menor temperatura: {menor:.2f} °C")
print(f"Temperatura média: {media:.2f} °C")
print(f"Dias acima da média: {dias_acima_media}")

print("====================================")
