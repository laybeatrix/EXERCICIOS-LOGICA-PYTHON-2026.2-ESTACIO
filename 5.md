while True:
    idade = int(input("Digite sua idade: "))

    if idade >= 0:
        break
    else:
        print("Erro: a idade não pode ser negativa. Digite novamente.")

if idade <= 12:
    classificacao = "Criança"
elif idade <= 17:
    classificacao = "Adolescente"
elif idade <= 59:
    classificacao = "Adulto"
else:
    classificacao = "Idoso"

print(f"\nClassificação etária: {classificacao}")
