def cadastrar_estudante():
    nome = input("Nome do estudante: ")

    notas = []

    for i in range(3):
        while True:
            nota = float(input(f"Digite a {i + 1}ª nota (0 a 10): "))

            if 0 <= nota <= 10:
                notas.append(nota)
                break
        else:
                print("Erro: a nota deve estar entre 0 e 10.")

    media = sum(notas) / 3

    estudante = {
        "nome": nome,
        "notas": notas,
        "media": media
    }

    return estudante


def exibir_resultados(estudantes):
    maior = max(estudantes, key=lambda estudante: estudante["media"])
    menor = min(estudantes, key=lambda estudante: estudante["media"])

    aprovados = 0
recuperacao = 0
    reprovados = 0

    print("\n===== MÉDIA DOS ESTUDANTES =====")

    for estudante in estudantes:
        print(f"{estudante['nome']}: {estudante['media']:.2f}")

        if estudante["media"] >= 7:
            aprovados += 1
        elif estudante["media"] >= 5:
            recuperacao += 1
    else:
            reprovados += 1

    print("\n===== DESTAQUES =====")
    print(f"Maior média: {maior['nome']} - {maior['media']:.2f}")
    print(f"Menor média: {menor['nome']} - {menor['media']:.2f}")

    print("\n===== SITUAÇÃO DA TIRMA =====")
    print(f"Aprovados: {aprovados}")
    print(f"Em recuperação: {recuperacao}")
    print(f"Reprovados: {reprovados}")


estudantes = []

print("===== CADASTRO DE ESTUDANTES =====")

for i in range(5):
    print(f"\n--- Estudante {i + 1} ---")
    estudante = cadastrar_estudante()
    estudantes.append(estudante)

exibir_resultados(estudantes)
