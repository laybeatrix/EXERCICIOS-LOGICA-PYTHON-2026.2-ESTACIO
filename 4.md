```python

notas = []

for i in range(3):
    while True:
        nota = float(input(f"Digite a {i + 1}ª nota (0 a 10): "))

        if 0 <= nota <= 10:
            notas.append(nota)
            break
        else:
            print("Erro: a nota deve estar entre 0 e 10. Digite novamente.")

media = sum(notas) / 3

if media >= 7:
    situacao = "Aprovado"
elif media >= 5:
    situacao = "Recuperação"
else:
    situacao = "Reprovado"

print("\n===== DESEMPENHO DO ESTUDANTE =====")
print(f"Nota 1: {notas[0]:.2f}")
print(f"Nota 2: {notas[1]:.2f}")
print(f"Nota 3: {notas[2]:.2f}")
print(f"Média: {media:.2f}")
print(f"Situação: {situacao}")
print("===================================")
