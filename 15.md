```python

cidades = []

print("===== CADASTRO DE CIDADES =====")

for i in range(5):
    print(f"\nCadastro da cidade {i + 1}")

    nome = input("Nome da cidade: ")
    estado = input("Estado (sigla): ").upper()
    populacao = int(input("População estimada: "))

    cidade = {
        "nome": nome,
        "estado": estado,
        "populacao": populacao
    }

    cidades.append(cidade)

# Cidade com maior população
maior = max(cidades, key=lambda cidade: cidade["populacao"])

# Cidade com menor população
menor = min(cidades, key=lambda cidade: cidade["populacao"])

# População total
populacao_total = sum(cidade["populacao"] for cidade in cidades)

# Média populacional
media = populacao_total / len(cidades)

print("\n===== RESULTADOS =====")

print(f"\nCidade com maior população:")
print(f"{maior['nome']} - {maior['estado']} ({maior['populacao']} habitantes)")

print(f"\nCidade com menor população:")
print(f"{menor['nome']} - {menor['estado']} ({menor['populacao']} habitantes)")

print(f"\nPopulação total: {populacao_total} habitantes")
print(f"Média populacional: {media:.2f} habitantes")

print("\n===== TODAS AS CIDADES =====")

for cidade in cidades:
    print(f"Cidade: {cidade['nome']}")
    print(f"Estado: {cidade['estado']}")
    print(f"População: {cidade['populacao']} habitantes")
    print("------------------------------")
