```python

import math

numero = float(input("Digite um número real: "))

if numero >= 0:
    print("Raiz quadrada:", math.sqrt(numero))
else:
    print("Raiz quadrada: não existe nos números reais.")

print("Valur absoluto:", math.fabs(numero))


print("Arrredondamento para cima (teto):", math.ceil(numero))


print("Arredondamento para baixo (piso):", math.floor(numero))

if numero.is_integer() and numero >= 0:
    print("Fatorial:", math.factorial(int(numero)))
else:
    print("Fatorial: não é possível calcular, pois o número não é um inteiro não negativo.")
