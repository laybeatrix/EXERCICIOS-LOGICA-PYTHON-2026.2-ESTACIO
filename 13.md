```python

contatos = []

print("===== AGENDA ELETRÔNICA =====")

for i in range(5):
    print(f"\n--- Cadastro do contato {i + 1} ---")

    nome = input("Nome: ")
    telefone = input("Telefone: ")
    email = input("E-mail: ")

    contato = {
        "nome": nome,
        "telefone": telefone,
        "email": email
    }

    contatos.append(contato)

print("\n===== CONSULTA DE CONTATO =====")

nome_busca = input("Digite o nome da pessoa que deseja consultar: ")

encontrado = False

for contato in contatos:
    if contato["nome"].lower() == nome_busca.lower():
        print("\nContato encontrado!")
        print(f"Nome: {contato['nome']}")
        print(f"Telefone: {contato['telefone']}")
        print(f"E-mail: {contato['email']}")

        encontrado = True
        break

if not encontrado:
    print("Contato não encontrado.")
